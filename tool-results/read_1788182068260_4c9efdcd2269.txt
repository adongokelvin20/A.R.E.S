   170→/* ============ Revenue chart ============ */
   171→function RevenueChart({ data }: { data: any }) {
   172→  const series: { date: string; revenue: number; orders: number }[] = data.series ?? [];
   173→  const gradientId = useId();
   174→  const width = 760, height = 220;
   175→  const padding = { top: 20, right: 16, bottom: 28, left: 48 };
   176→  const innerW = width - padding.left - padding.right;
   177→  const innerH = height - padding.top - padding.bottom;
   178→  const maxRevenue = Math.max(...series.map((s) => s.revenue), 1);
   179→  const points = series.map((s, i) => ({
   180→    x: padding.left + (i / Math.max(series.length - 1, 1)) * innerW,
   181→    y: padding.top + innerH - (s.revenue / maxRevenue) * innerH,
   182→    ...s,
   183→  }));
   184→  const linePath = points.map((p, i) => `${i === 0 ? "M" : "L"} ${p.x.toFixed(2)} ${p.y.toFixed(2)}`).join(" ");
   185→  const areaPath = `M ${points[0]?.x ?? 0} ${padding.top + innerH} ` + points.map((p) => `L ${p.x.toFixed(2)} ${p.y.toFixed(2)}`).join(" ") + ` L ${points[points.length - 1]?.x ?? 0} ${padding.top + innerH} Z`;
   186→  const total = series.reduce((s, p) => s + p.revenue, 0);
   187→
   188→  if (series.length === 0 || total === 0) {
   189→    return (
   190→      <div className="rounded-2xl border border-ares-line bg-white p-5">
   191→        <div className="flex items-center gap-2">
   192→          <BarChart3 className="h-4 w-4 text-ares-sea-deep" />
   193→          <div className="text-sm font-semibold text-ares-navy">Revenue · last 14 days</div>
   194→        </div>
   195→        <div className="flex h-48 items-center justify-center text-xs text-muted-foreground">No revenue yet. Orders will populate this chart.</div>
   196→      </div>
   197→    );
   198→  }
   199→
   200→  return (
   201→    <div className="rounded-2xl border border-ares-line bg-white p-5">
   202→      <div className="flex items-start justify-between">
   203→        <div>
   204→          <div className="flex items-center gap-2">
   205→            <BarChart3 className="h-4 w-4 text-ares-sea-deep" />
   206→            <div className="text-sm font-semibold text-ares-navy">Revenue · last 14 days</div>
   207→          </div>
   208→          <div className="mt-1 font-mono text-2xl font-bold text-ares-navy">
   209→            {data.business.currency === "GHS" ? "GH₵" : data.business.currency} {total.toLocaleString(undefined, { maximumFractionDigits: 0 })}
   210→          </div>
   211→        </div>
   212→        <span className="inline-flex items-center gap-1.5 rounded-md bg-ares-foam px-2 py-1 text-xs font-medium text-ares-sea-deep">
   213→          <span className="h-2 w-2 rounded-full bg-ares-sea" /> Revenue
   214→        </span>
   215→      </div>
   216→      <div className="mt-3">
   217→        <svg viewBox={`0 0 ${width} ${height}`} className="h-auto w-full" preserveAspectRatio="none">
   218→          <defs>
   219→            <linearGradient id={gradientId} x1="0" y1="0" x2="0" y2="1">
   220→              <stop offset="0%" stopColor="#0284A6" stopOpacity="0.25" />
   221→              <stop offset="100%" stopColor="#0284A6" stopOpacity="0" />
   222→            </linearGradient>
   223→          </defs>
   224→          {[0, 0.25, 0.5, 0.75, 1].map((t) => {
   225→            const y = padding.top + t * innerH;
   226→            return <line key={t} x1={padding.left} y1={y} x2={padding.left + innerW} y2={y} stroke="#E4ECF1" strokeDasharray="3 3" />;
   227→          })}
   228→          {[0, 0.5, 1].map((t) => {
   229→            const y = padding.top + (1 - t) * innerH;
   230→            return <text key={t} x={padding.left - 8} y={y + 3} fontSize="9" fill="#5C6B7A" textAnchor="end" fontFamily="monospace">{Math.round(t * maxRevenue)}</text>;
   231→          })}
   232→          <path d={areaPath} fill={`url(#${gradientId})`} />
   233→          <path d={linePath} fill="none" stroke="#024E6E" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   234→          {points.map((p, i) => <circle key={i} cx={p.x} cy={p.y} r="3" fill="#0284A6" stroke="white" strokeWidth="1.5" />)}
   235→          {points.map((p, i) => i % 2 === 0 ? <text key={i} x={p.x} y={height - 8} fontSize="9" fill="#5C6B7A" textAnchor="middle" fontFamily="monospace">{p.date}</text> : null)}
   236→        </svg>
   237→      </div>
   238→    </div>
   239→  );
   240→}

   242→/* ============ Channel donut ============ */
   243→function ChannelDonut({ data }: { data: any }) {
   244→  const channels: { name: string; value: number }[] = data.channelBreakdown ?? [];
   245→  const total = channels.reduce((s, c) => s + c.value, 0) || 1;
   246→  const radius = 60;
   247→  const circumference = 2 * Math.PI * radius;
   248→  const colors: Record<string, string> = { WHATSAPP: "#25D366", WEB: "#0284A6", INSTORE: "#7AB8CC", CALL: "#0A1626", SMS: "#38BDF8", EMAIL: "#1A5694" };
   249→  const segments = channels.map((c, i) => {
   250→    const pct = c.value / total;
   251→    const dash = pct * circumference;
   252→    const offset = channels.slice(0, i).reduce((s, x) => s + (x.value / total) * circumference, 0);
   253→    return { ...c, dash, offset, color: colors[c.name] ?? "#0284A6" };
   254→  });
   255→
   256→  return (
   257→    <div className="rounded-2xl border border-ares-line bg-white p-5">
   258→      <div className="flex items-center gap-2"><PieChart className="h-4 w-4 text-ares-sea-deep" /><div className="text-sm font-semibold text-ares-navy">Orders by channel</div></div>
   259→      {channels.length === 0 ? (
   260→        <div className="flex h-48 items-center justify-center text-xs text-muted-foreground">No orders yet.</div>
   261→      ) : (
   262→        <div className="mt-3 flex items-center gap-5">
   263→          <div className="relative h-32 w-32 shrink-0">
   264→            <svg viewBox="0 0 160 160" className="h-full w-full -rotate-90">
   265→              {segments.map((s, i) => <circle key={i} cx="80" cy="80" r={radius} fill="none" stroke={s.color} strokeWidth="14" strokeDasharray={`${s.dash} ${circumference - s.dash}`} strokeDashoffset={-s.offset} />)}
   266→            </svg>
   267→            <div className="absolute inset-0 flex flex-col items-center justify-center">
   268→              <div className="font-mono text-2xl font-bold text-ares-navy">{total}</div>
   269→              <div className="text-[10px] uppercase tracking-wide text-muted-foreground">orders</div>
   270→            </div>
   271→          </div>
   272→          <div className="flex-1 space-y-2">
   273→            {channels.map((c) => (
   274→              <div key={c.name} className="flex items-center justify-between text-xs">
   275→                <div className="flex items-center gap-2">
   276→                  <span className="h-2.5 w-2.5 rounded-sm" style={{ background: colors[c.name] ?? "#0284A6" }} />
   277→                  <span className="font-medium text-ares-navy">{c.name}</span>
   278→                </div>
   279→                <div className="flex items-center gap-2">
   280→                  <span className="text-muted-foreground">{Math.round((c.value / total) * 100)}%</span>
   281→                  <span className="font-mono font-semibold text-ares-navy">{c.value}</span>
   282→                </div>
   283→              </div>
   284→            ))}
   285→          </div>
   286→        </div>
   287→      )}
   288→    </div>
   289→  );
   290→}

   456→/* ============ Order status pie chart ============ */
   457→function OrderStatusPie({ data }: { data: any }) {
   458→  const statuses: { name: string; value: number }[] = data.statusBreakdown ?? [];
   459→  const total = statuses.reduce((s, c) => s + c.value, 0) || 1;
   460→  const colors: Record<string, string> = {
   461→    PENDING: "#F59E0B",
   462→    CONFIRMED: "#0284A6",
   463→    FULFILLED: "#10B981",
   464→    CANCELLED: "#94A3B8",
   465→    REFUNDED: "#EF4444",
   466→  };
   467→  const labels: Record<string, string> = {
   468→    PENDING: "Pending",
   469→    CONFIRMED: "Confirmed",
   470→    FULFILLED: "Closed",
   471→    CANCELLED: "Cancelled",
   472→    REFUNDED: "Refunded",
   473→  };
   474→  const radius = 60;
   475→  const circumference = 2 * Math.PI * radius;
   476→  const segments = statuses.map((c, i) => {
   477→    const pct = c.value / total;
   478→    const dash = pct * circumference;
   479→    const offset = statuses.slice(0, i).reduce((s, x) => s + (x.value / total) * circumference, 0);
   480→    return { ...c, dash, offset, color: colors[c.name] ?? "#0284A6" };
   481→  });
   482→
   483→  return (
   484→    <div className="rounded-2xl border border-ares-line bg-white p-5">
   485→      <div className="flex items-center gap-2">
   486→        <PieChart className="h-4 w-4 text-ares-sea-deep" />
   487→        <div className="text-sm font-semibold text-ares-navy">Order status</div>
   488→      </div>
   489→      {statuses.length === 0 ? (
   490→        <div className="flex h-48 items-center justify-center text-xs text-muted-foreground">No orders yet.</div>
   491→      ) : (
   492→        <div className="mt-3 flex items-center gap-5">
   493→          <div className="relative h-32 w-32 shrink-0">
   494→            <svg viewBox="0 0 160 160" className="h-full w-full -rotate-90">
   495→              {segments.map((s, i) => (
   496→                <circle key={i} cx="80" cy="80" r={radius} fill="none" stroke={s.color} strokeWidth="14" strokeDasharray={`${s.dash} ${circumference - s.dash}`} strokeDashoffset={-s.offset} />
   497→              ))}
   498→            </svg>
   499→            <div className="absolute inset-0 flex flex-col items-center justify-center">
   500→              <div className="font-mono text-2xl font-bold text-ares-navy">{total}</div>
   501→              <div className="text-[10px] uppercase tracking-wide text-muted-foreground">orders</div>
   502→            </div>
   503→          </div>
   504→          <div className="flex-1 space-y-2">
   505→            {statuses.map((s) => (
   506→              <div key={s.name} className="flex items-center justify-between text-xs">
   507→                <div className="flex items-center gap-2">
   508→                  <span className="h-2.5 w-2.5 rounded-full" style={{ background: colors[s.name] ?? "#0284A6" }} />
   509→                  <span className="font-medium text-ares-navy">{labels[s.name] ?? s.name}</span>
   510→                </div>
   511→                <div className="flex items-center gap-2">
   512→                  <span className="text-muted-foreground">{Math.round((s.value / total) * 100)}%</span>
   513→                  <span className="font-mono font-semibold text-ares-navy">{s.value}</span>
   514→                </div>
   515→              </div>
   516→            ))}
   517→          </div>
   518→        </div>
   519→      )}
   520→    </div>
   521→  );
   522→}
